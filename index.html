Comprueba la dirección.`); } } // ═══════════════════════════════════════════════════════════════════════════ // 4. `; GmailApp.sendEmail(email, asunto, '', { from: CONFIG.emailRemitente, name: CONFIG.nombreRemitente, htmlBody: cuerpoHtml, }); return true; } catch (err) { console.error('Error enviando email de confirmación:', err); return false; } } // ═══════════════════════════════════════════════════════════════════════════ // 5. MOVER FILA DE PENDIENTES → ALUMNOS ACTIVOS // ═══════════════════════════════════════════════════════════════════════════ function moverFilaAActivos(hojaPendientes, fila, nombre, email, clases, horas, precioBase) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const hojaAlumnos = ss.getSheetByName(CONFIG.hojaAlumnos); const hojaPagos = ss.getSheetByName(CONFIG.hojaPagos); const nuevoId = hojaAlumnos.getLastRow() - 3; const hojaPend = hojaPendientes || ss.getSheetByName(CONFIG.hojaPendientes); const edad = hojaPend.getRange(fila, 5).getValue(); hojaAlumnos.appendRow([ nuevoId, nombre, email, hojaPend.getRange(fila, 4).getValue(), // teléfono new Date(), edad, clases, horas, precioBase, 'Pagado', new Date(), 'Sí', 'Transferencia', '', ]); const mesActual = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM'); hojaPagos.appendRow([ nuevoId, nombre, mesActual, precioBase, new Date(), 'Transferencia', 'Pagado', 0, precioBase, ]); // Tachar la fila en Pendientes para que se vea visualmente que está procesada hojaPend.getRange(fila, 1, 1, hojaPend.getLastColumn()) .setFontLine('line-through') .setFontColor('#888888'); } catch (err) { console.error('Error moviendo fila a Activos:', err); } } // ═══════════════════════════════════════════════════════════════════════════ // 6. Comprueba la dirección: ${email}`); } } function moverConfirmadoAActivos() { const hoja = SpreadsheetApp.getActiveSheet(); if (hoja.getName() !== CONFIG.hojaPendientes) { SpreadsheetApp.getUi().alert('Selecciona una fila en "Pendientes Transferencia".'); return; } const fila = SpreadsheetApp.getActiveRange().getRow(); const nombre = hoja.getRange(fila, CONFIG.colNombre).getValue(); const email = hoja.getRange(fila, CONFIG.colEmail).getValue(); const clases = hoja.getRange(fila, CONFIG.colClases).getValue(); const horas = hoja.getRange(fila, CONFIG.colHoras).getValue(); const precio = hoja.getRange(fila, CONFIG.colPrecioBase).getValue(); moverFilaAActivos(hoja, fila, nombre, email, clases, horas, precio); SpreadsheetApp.getUi().alert(`✅ ${nombre} movido/a a Alumnos Activos.`); } function comprobarPendientes() { const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.hojaPendientes); const datos = hoja.getDataRange().getValues(); let pendientes = 0; for (let i = 1; i < datos.length; i++) { const estado = datos[i][CONFIG.colEstado - 1]; if (estado === 'Pendiente de pago') pendientes++; } SpreadsheetApp.getUi().alert(`Hay ${pendientes} transferencia(s) pendientes de confirmar.`); }
